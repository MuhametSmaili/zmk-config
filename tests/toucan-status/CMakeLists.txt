cmake_minimum_required(VERSION 3.20)
project(toucan_status_preview C)
if(NOT LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the LVGL source used by the firmware")
endif()
unset(LV_CONF_PATH CACHE)
set(LV_CONF_INCLUDE_SIMPLE ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_SOURCE_DIR} lvgl)
target_include_directories(lvgl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(screen-preview screen-preview.c ../../boards/shields/toucan_display/screen.c)
target_include_directories(screen-preview PRIVATE ../../include)
target_link_libraries(screen-preview PRIVATE lvgl m)
